app.js JS
const atc = document.querySelector(
  '#atc'
);

function order() {
  atc.classList.add('is-packing');
  // cart packs, truck drives
  setTimeout(() => {
    document.body.classList.add(
      'is-rolling'
    );
  }, 650);
  setTimeout(() => {
    document.body.classList.add(
      'is-ordered'
    );
  }, 2300);
}
